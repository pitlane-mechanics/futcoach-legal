  try{
    var c=(new URLSearchParams(location.search).get("c")||"").toLowerCase().replace(/[^a-z0-9_-]/g,"").slice(0,30);
    if(!c)return;
    document.querySelectorAll("a[data-store]").forEach(function(a){a.href=a.href.replace("ct=web","ct="+c)});
  }catch(e){}
})();
